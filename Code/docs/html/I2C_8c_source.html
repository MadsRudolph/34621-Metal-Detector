60void I2C_Init(void)
61{
62 /*
63 * FJERNET: Startup pin kode
64 *
65 * ORIGINAL KODE:
66 * #if I2C_HAS_STARTUP_PIN
67 * I2C_STARTUP_DDR |= (1 << I2C_STARTUP_BIT);
68 * I2C_STARTUP_PORT |= (1 << I2C_STARTUP_BIT);
69 * #endif
70 *
71 * HVORFOR FJERNET:
72 * - I2C_STARTUP_PIN (PA0) findes ikke på ATmega328P
73 * - Nano har ikke Port A overhovedet
74 */
75
76 _delay_ms(1000); /* Startup delay for display */
77
78 /*
79 * TWI Configuration:
80 * SCL Frequency: SCL = F_CPU / (16 + 2 * TWBR * Prescaler)
81 *
82 * Med TWBR = 18, Prescaler = 1:
83 * SCL = 16000000 / (16 + 2 * 18 * 1) = 16000000 / 52 = ~308 kHz
84 *
85 * Datasheet: Section 21.5.2, Page 180 - Bit Rate Generator
86 * Datasheet: Section 21.9.1, Page 198 - TWBR register
87 */
88 TWBR = 18;
89
90 /*
91 * Clear prescaler bits (prescaler = 1)
92 * TWPS1:0 = 00 gives prescaler = 1
93 * Datasheet: Section 21.9.3, Page 200 - TWSR register
94 * Datasheet: Table 21-8, Page 200 - TWI Bit Rate Prescaler
95 */
96 TWSR &= 0xFC;
97
98 /*
99 * Enable TWI
100 * TWCR = 0x05 = TWEN (bit 2) + TWIE (bit 0)
101 * Datasheet: Section 21.9.2, Page 199 - TWCR register
102 */
103 TWCR = 0x05;
104}

119uint8_t I2C_Start(char write_address)
120{
121 uint8_t status;
122
123 /*
124 * Generate START condition
125 * TWSTA = 1: Generate START
126 * TWEN = 1: Enable TWI
127 * TWINT = 1: Clear interrupt flag (starts operation)
128 * Datasheet: Section 21.9.2, Page 199 - TWCR register
129 */
130 TWCR = (1 << TWSTA) | (1 << TWEN) | (1 << TWINT);
131
132 /* Wait for TWI to complete (TWINT becomes 1) */
133 while (!(TWCR & (1 << TWINT)));
134
135 /*
136 * Check status (mask prescaler bits)
137 * 0x08 = START transmitted
138 * Datasheet: Table 21-3, Page 186 - Status code 0x08
139 */
140 status = TWSR & 0xF8;
141 if (status != 0x08) {
142 return 0;
143 }
144
145 /*
146 * Send SLA+W (slave address + write)
147 * Datasheet: Section 21.9.4, Page 200 - TWDR register
148 */
149 TWDR = write_address;
150 TWCR = (1 << TWEN) | (1 << TWINT);
151
152 /* Wait for TWI to complete */
153 while (!(TWCR & (1 << TWINT)));
154
155 /*
156 * Check status
157 * 0x18 = SLA+W transmitted, ACK received
158 * 0x20 = SLA+W transmitted, NACK received
159 * Datasheet: Table 21-3, Page 186 - Status codes 0x18, 0x20
160 */
161 status = TWSR & 0xF8;
162 if (status == 0x18) {
163 return 1;
164 }
165 if (status == 0x20) {
166 return 2;
167 }
168
169 return 3;
170}

183uint8_t I2C_Repeated_Start(char read_address)
184{
185 uint8_t status;
186
187 /* Generate repeated START */
188 TWCR = (1 << TWSTA) | (1 << TWEN) | (1 << TWINT);
189
190 /* Wait for TWI to complete */
191 while (!(TWCR & (1 << TWINT)));
192
193 /* Check status */
194 status = TWSR & 0xF8;
195 if (status != 0x10) { /* Repeated START not transmitted */
196 return 0;
197 }
198
199 /* Send SLA+R (slave address + read) */
200 TWDR = read_address;
201 TWCR = (1 << TWEN) | (1 << TWINT);
202
203 /* Wait for TWI to complete */
204 while (!(TWCR & (1 << TWINT)));
205
206 /* Check status */
207 status = TWSR & 0xF8;
208 if (status == 0x40) { /* SLA+R transmitted, ACK received */
209 return 1;
210 }
211 if (status == 0x20) { /* SLA+R transmitted, NACK received */
212 return 2;
213 }
214
215 return 3; /* SLA+R failed */
216}

228uint8_t I2C_Write(char data)
229{
230 uint8_t status;
231
232 /* Load data */
233 TWDR = data;
234 TWCR = (1 << TWEN) | (1 << TWINT);
235
236 /* Wait for TWI to complete */
237 while (!(TWCR & (1 << TWINT)));
238
239 /* Check status */
240 status = TWSR & 0xF8;
241 if (status == 0x28) { /* Data transmitted, ACK received */
242 return 0;
243 }
244 if (status == 0x30) { /* Data transmitted, NACK received */
245 return 1;
246 }
247
248 return 2; /* Transmission failure */
249}
