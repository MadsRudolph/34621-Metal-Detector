367void dft_sum(int16_t raw) {
368 int16_t x = raw - ADC_OFFSET; /* Remove DC offset */
369
370#if CAPTURE_SAMPLES
371 uint8_t state = capture_get_state();
372 if (dft_index == 0 && state == CAPTURE_WAITING) {
373 capture_set_state(CAPTURE_ACTIVE);
374 }
375 capture_store_sample(dft_index, x);
376#endif
377
378 /* Apply DFT coefficients based on sample position */
379 switch (dft_index & 3) {
380 case 0: Re += x; break; /* cos(0°) = +1 */
381 case 1: Im -= x; break; /* sin(90°) = +1, but we use -sin */
382 case 2: Re -= x; break; /* cos(180°) = -1 */
383 case 3: Im += x; break; /* sin(270°) = -1, but we use -sin */
384 }
385
386 if (++dft_index >= N) {
387 /* Window complete - buffer results and reset */
388 Re_buf = Re;
389 Im_buf = Im;
390 Re = Im = 0;
391 dft_index = 0;
392 dft_done = 1;
393 sync_flag = 0;
394
395#if CAPTURE_SAMPLES
396 if (capture_get_state() == CAPTURE_ACTIVE) {
397 capture_set_state(CAPTURE_DONE);
398 }
399#endif
400 }

413void dft_calc(void) {
414 double re = Re_buf, im = Im_buf;
415 mag = sqrt(re*re + im*im) / N;
416 phase = atan2(im, re) * 57.2957795; /* 180/π ≈ 57.2957795 */
417}

439void filter_update(void) {
440 mag_filtered = IIR_ALPHA * mag + (1.0f - IIR_ALPHA) * mag_filtered;
441 phase_filtered = IIR_ALPHA * phase + (1.0f - IIR_ALPHA) * phase_filtered;
442}

558void classify_metal(void) {
559 if (!is_calibrated) {
560 metal_type = METAL_NONE;
561 metal_detected = 0;
562 return;
563 }
564
565 /* Calculate absolute magnitude change */
566 int16_t delta_mag = (int16_t)mag_filtered - (int16_t)baseline_mag;
567 if (delta_mag < 0) delta_mag = -delta_mag;
568
569 /* Check detection threshold */
570 if (delta_mag < DETECT_THRESHOLD) {
571 metal_detected = 0;
572 metal_type = METAL_NONE;
573 return;
574 }
575
576 /* Metal detected - classify by phase shift */
577 metal_detected = 1;
578 int16_t delta_phase = (int16_t)phase_filtered - baseline_phase;
579 if (delta_phase < FERRO_THRESHOLD) {
580 metal_type = METAL_FERRO;
581 } else {
582 metal_type = METAL_NONFERRO;
583 }
584}
