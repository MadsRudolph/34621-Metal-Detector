529// dim = false: display is normal
530void dim(bool dim) {
531 uint8_t contrast;
532
533 if (dim) {
534 contrast = 0; // Dimmed display
535 } else {
536 if (_vccstate == SSD1306_EXTERNALVCC) {
537 contrast = 0x9F;
538 } else {
539 contrast = 0xCF;
540 }
541 }
542 // the range of contrast to too small to be really useful
543 // it is useful to dim the display
544 ssd1306_command(SSD1306_SETCONTRAST);
545 ssd1306_command(contrast);
546}
