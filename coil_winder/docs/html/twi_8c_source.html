15void twi_init(void)
16{
17 /*
18 * Set prescaler to 1 (TWPS = 0 in TWSR)
19 * TWSR bits 1:0 are TWPS1:TWPS0
20 */
21 TWSR = 0;
22
23 /*
24 * Set bit rate for desired frequency:
25 * SCL_freq = F_CPU / (16 + 2 * TWBR * prescaler)
26 * TWBR = ((F_CPU / SCL_freq) - 16) / (2 * prescaler)
27 *
28 * At 16MHz, 400kHz, prescaler=1:
29 * TWBR = ((16000000 / 400000) - 16) / 2 = (40 - 16) / 2 = 12
30 */
31 TWBR = ((F_CPU / TWI_FREQ) - 16) / 2;
32
33 /* Enable TWI peripheral */
34 TWCR = (1 << TWEN);
35}

37void twi_start(void)
38{
39 /*
40 * Send START condition:
41 * - TWINT: Clear interrupt flag (by writing 1)
42 * - TWSTA: Generate START condition
43 * - TWEN: Keep TWI enabled
44 */
45 TWCR = (1 << TWINT) | (1 << TWSTA) | (1 << TWEN);
46
47 /* Wait for START to be transmitted */
48 while (!(TWCR & (1 << TWINT)));
49}

51void twi_stop(void)
52{
53 /*
54 * Send STOP condition:
55 * - TWINT: Clear interrupt flag
56 * - TWSTO: Generate STOP condition
57 * - TWEN: Keep TWI enabled
58 *
59 * Note: TWINT is not set after STOP, so no wait needed
60 */
61 TWCR = (1 << TWINT) | (1 << TWSTO) | (1 << TWEN);
62}
