25void buzzer_beep(uint16_t freq_hz, uint16_t duration_ms)
26{
27 /*
28 * Calculate half-period in microseconds:
29 * half_period = 1000000 / (2 * freq) = 500000 / freq
30 */
31 uint16_t half_period = 500000UL / freq_hz;
32
33 /*
34 * Calculate number of complete cycles:
35 * cycles = freq * duration / 1000
36 */
37 uint16_t cycles = ((uint32_t)freq_hz * duration_ms) / 1000;
38
39 /* Generate square wave */
40 for (uint16_t i = 0; i < cycles; i++) {
41 /* High phase */
42 BUZZER_PORT |= (1 << BUZZER_PIN);
43 for (uint16_t j = 0; j < half_period; j++) {
44 _delay_us(1);
45 }
46
47 /* Low phase */
48 BUZZER_PORT &= ~(1 << BUZZER_PIN);
49 for (uint16_t j = 0; j < half_period; j++) {
50 _delay_us(1);
51 }
52 }
53}
