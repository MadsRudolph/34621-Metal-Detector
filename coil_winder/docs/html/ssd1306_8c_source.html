129static void ssd1306_cmd(uint8_t cmd)
130{
131 twi_start();
132 twi_write(SSD1306_ADDR << 1); /* Address + write bit */
133 twi_write(0x00); /* Control byte: command mode */
134 twi_write(cmd);
135 twi_stop();
136}

176void ssd1306_update(void)
177{
178 /* Set column address range (0-127) */
179 ssd1306_cmd(0x21);
180 ssd1306_cmd(0);
181 ssd1306_cmd(127);
182
183 /* Set page address range (0-7) */
184 ssd1306_cmd(0x22);
185 ssd1306_cmd(0);
186 ssd1306_cmd(7);
187
188 /* Send framebuffer data */
189 twi_start();
190 twi_write(SSD1306_ADDR << 1); /* Address + write bit */
191 twi_write(0x40); /* Control byte: data mode */
192
193 for (uint16_t i = 0; i < sizeof(g_display_buffer); i++) {
194 twi_write(g_display_buffer[i]);
195 }
196
197 twi_stop();
198}

200void ssd1306_set_pixel(uint8_t x, uint8_t y, bool on)
201{
202 /* Bounds check */
203 if (x >= OLED_WIDTH || y >= OLED_HEIGHT) {
204 return;
205 }
206
207 /* Calculate buffer index and bit position */
208 uint16_t idx = x + (y / 8) * OLED_WIDTH;
209 uint8_t bit = y % 8;
210
211 if (on) {
212 g_display_buffer[idx] |= (1 << bit);
213 } else {
214 g_display_buffer[idx] &= ~(1 << bit);
215 }
216}

218void ssd1306_draw_char(uint8_t x, uint8_t y, char c, uint8_t scale)
219{
220 /* Clamp character to valid range */
221 if (c < 32 || c > 122) {
222 c = '?';
223 }
224
225 uint8_t idx = c - 32;
226
227 /* Draw each column of the character */
228 for (uint8_t col = 0; col < 5; col++) {
229 uint8_t line = font5x7[idx * 5 + col];
230
231 /* Draw each row of the column */
232 for (uint8_t row = 0; row < 7; row++) {
233 bool pixel = (line >> row) & 1;
234
235 /* Scale the pixel */
236 for (uint8_t sy = 0; sy < scale; sy++) {
237 for (uint8_t sx = 0; sx < scale; sx++) {
238 ssd1306_set_pixel(x + col * scale + sx,
239 y + row * scale + sy,
240 pixel);
241 }
242 }
243 }
244 }
245}

255void ssd1306_draw_int(uint8_t x, uint8_t y, int16_t num, uint8_t digits, uint8_t scale)
256{
257 char buf[8];
258 int8_t i = digits - 1;
259
260 /* Convert to positive for digit extraction */
261 if (num < 0) {
262 num = -num;
263 }
264
265 /* Build string right-to-left (zero-padded) */
266 buf[digits] = '\0';
267 while (i >= 0) {
268 buf[i--] = '0' + (num % 10);
269 num /= 10;
270 }
271
272 ssd1306_draw_string(x, y, buf, scale);
273}

275void ssd1306_draw_rect(uint8_t x, uint8_t y, uint8_t w, uint8_t h, bool fill)
276{
277 for (uint8_t i = x; i < x + w; i++) {
278 for (uint8_t j = y; j < y + h; j++) {
279 /* Draw if filled, or if on edge */
280 if (fill || i == x || i == x + w - 1 || j == y || j == y + h - 1) {
281 ssd1306_set_pixel(i, j, true);
282 }
283 }
284 }
285}

287void ssd1306_draw_progress(uint8_t x, uint8_t y, uint8_t w, uint8_t h,
288 uint16_t value, uint16_t max_val)
289{
290 /* Draw outline */
291 ssd1306_draw_rect(x, y, w, h, false);
292
293 /* Calculate and draw fill */
294 if (max_val > 0 && value > 0) {
295 uint8_t fill = ((uint32_t)(w - 2) * value) / max_val;
296
297 /* Clamp to available width */
298 if (fill > w - 2) {
299 fill = w - 2;
300 }
301
302 if (fill > 0) {
303 ssd1306_draw_rect(x + 1, y + 1, fill, h - 2, true);
304 }
305 }
