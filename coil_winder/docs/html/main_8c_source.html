59static void process_turn(void)
60{
61 const coil_preset_t *preset = preset_get_current(&g_winder);
62
63 /* Move traverse by wire diameter (in mm * 100) */
64 int16_t step = preset->wire_diameter;
65
66 if (g_winder.traverse_direction) {
67 g_winder.position_x100 += step;
68 } else {
69 g_winder.position_x100 -= step;
70 }
71
72 /* Check boundaries and handle layer change */
73 int16_t max_pos = (int16_t)preset->winding_width * 100;
74
75 if (g_winder.position_x100 >= max_pos) {
76 /* Hit right boundary - reverse direction */
77 g_winder.position_x100 = max_pos;
78 g_winder.traverse_direction = false;
79 g_winder.current_layer++;
80 } else if (g_winder.position_x100 <= 0) {
81 /* Hit left boundary - reverse direction */
82 g_winder.position_x100 = 0;
83 g_winder.traverse_direction = true;
84 g_winder.current_layer++;
85 }
86
87 /* Update servo to new position */
88 servo_set_position_mm(g_winder.position_x100, preset->winding_width);
89
90 /* Check for winding completion */
91 if (encoder_get_count() >= preset->total_turns) {
92 g_winder.state = STATE_COMPLETE;
93 buzzer_complete_melody();
94 }
95}

163static void handle_buttons(void)
164{
165 uint8_t events = buttons_poll();
166
167 /* Reset button - always active */
168 if (events & BTN_RESET_PRESSED) {
169 winder_reset(&g_winder);
170 }
171
172 /* Preset button - only when idle or complete */
173 if (events & BTN_PRESET_PRESSED) {
174 if (g_winder.state == STATE_IDLE || g_winder.state == STATE_COMPLETE) {
175 preset_next(&g_winder);
176 winder_reset(&g_winder);
177 }
178 }
179
180 /* Start/Stop button - state-dependent action */
181 if (events & BTN_START_PRESSED) {
182 if (g_winder.state == STATE_IDLE || g_winder.state == STATE_PAUSED) {
183 /* Start or resume winding */
184 g_winder.state = STATE_WINDING;
185 } else if (g_winder.state == STATE_WINDING) {
186 /* Pause winding */
187 g_winder.state = STATE_PAUSED;
188 } else if (g_winder.state == STATE_COMPLETE) {
189 /* Reset after completion */
190 winder_reset(&g_winder);
191 }
192 }
193}

206int main(void)
207{
208 /* Initialize all peripherals */
209 timer_init(); /* Millisecond timer (Timer0) */
210 twi_init(); /* I2C for OLED */
211 encoder_init(); /* Turn counter (INT0) */
212 servo_init(); /* Wire guide (Timer1 PWM) */
213 buttons_init(); /* User input */
214 buzzer_init(); /* Audio feedback */
215 ssd1306_init(); /* OLED display */
216
217 /* Enable global interrupts */
218 sei();
219
220 /* Reset to initial state */
221 winder_reset(&g_winder);
222
223 /* Startup beep to indicate ready */
224 buzzer_beep(1000, 100);
225
226 /* Display update timing */
227 uint32_t last_display = 0;
228
229 /* Main loop */
230 while (1) {
231 /* Handle button input */
232 handle_buttons();
233
234 /* Process turns when winding (with debounce filtering) */
235 if (encoder_check_turn() && g_winder.state == STATE_WINDING) {
236 process_turn();
237 }
238
239 /* Update display at 10Hz (100ms interval) */
240 if (millis() - last_display >= DISPLAY_UPDATE_MS) {
241 last_display = millis();
242 update_display();
243 }
244 }
245
246 return 0;
247}
