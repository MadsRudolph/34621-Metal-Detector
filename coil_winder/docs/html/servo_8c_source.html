15void servo_init(void)
16{
17 /* Configure PB1 (OC1A) as output */
18 SERVO_DDR |= (1 << SERVO_PIN);
19
20 /*
21 * Timer1 Fast PWM Mode 14 (WGM13:10 = 1110)
22 * - TOP = ICR1
23 * - Non-inverting output on OC1A (COM1A1:0 = 10)
24 * - Prescaler 8 (CS11)
25 */
26 TCCR1A = (1 << COM1A1) | (1 << WGM11);
27 TCCR1B = (1 << WGM13) | (1 << WGM12) | (1 << CS11);
28
29 /*
30 * Period calculation:
31 * f_timer = 16MHz / 8 = 2MHz
32 * Period = (ICR1 + 1) / 2MHz = 40000 / 2MHz = 20ms = 50Hz
33 */
34 ICR1 = 39999;
35
36 /*
37 * Center position (1.5ms pulse):
38 * 1.5ms * 2MHz = 3000 counts
39 */
40 OCR1A = 3000;
41}

43void servo_set_angle(uint8_t angle)
44{
45 /* Clamp angle to valid range */
46 if (angle > 180) {
47 angle = 180;
48 }
49
50 /*
51 * Map angle to pulse width:
52 * 0 degrees = 1.0ms = 2000 counts
53 * 180 degrees = 2.0ms = 4000 counts
54 *
55 * pulse = 2000 + (angle * 2000 / 180)
56 */
57 uint16_t pulse = 2000 + ((uint32_t)angle * 2000 / 180);
58 OCR1A = pulse;
59}

61void servo_set_position_mm(uint16_t position_x100, uint8_t width_mm)
62{
63 /*
64 * Map linear position to servo angle:
65 * position = 0 -> angle = CENTER - RANGE (e.g., 65 degrees)
66 * position = width_mm -> angle = CENTER + RANGE (e.g., 115 degrees)
67 *
68 * angle_offset = (position / width) * RANGE * 2 - RANGE
69 * angle = CENTER + angle_offset
70 */
71 uint16_t width_x100 = (uint16_t)width_mm * 100;
72
73 int16_t angle_offset = ((int32_t)position_x100 * SERVO_RANGE * 2 / width_x100) - SERVO_RANGE;
74 int16_t angle = SERVO_CENTER + angle_offset;
75
76 /* Clamp to valid servo range */
77 if (angle < 0) {
78 angle = 0;
79 }
80 if (angle > 180) {
81 angle = 180;
82 }
83
84 servo_set_angle((uint8_t)angle);
85}
