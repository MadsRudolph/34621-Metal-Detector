67bool encoder_check_turn(void)
68{
69 /* No new turn pending */
70 if (!g_new_turn) {
71 return false;
72 }
73
74 uint32_t now = millis();
75
76 /* Check if trigger is within debounce window (likely noise) */
77 if (now - g_last_trigger < ENCODER_DEBOUNCE_MS) {
78 /* Too fast - filter as noise */
79 g_new_turn = false;
80
81 /* Decrement the count that was erroneously incremented by noise */
82 cli();
83 if (g_turn_count > 0) {
84 g_turn_count--;
85 }
86 sei();
87
88 return false;
89 }
90
91 /* Valid turn - update timestamp and clear flag */
92 g_last_trigger = now;
93 g_new_turn = false;
94
95 return true;
96}
