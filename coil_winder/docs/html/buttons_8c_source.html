36uint8_t buttons_poll(void)
37{
38 uint8_t events = 0;
39
40 /* Read current button state (masked to only our pins) */
41 uint8_t btn_state = BTN_PINR & ((1 << BTN_RESET_PIN) | (1 << BTN_PRESET_PIN) | (1 << BTN_START_PIN));
42 uint32_t now = millis();
43
44 /*
45 * Debounce check:
46 * If we're within the debounce window, don't report events
47 * but still update the previous state to track releases.
48 * This prevents the button from appearing "stuck" if released
49 * during the debounce period.
50 */
51 if (now - g_last_time < DEBOUNCE_MS) {
52 g_btn_prev = btn_state;
53 return 0;
54 }
55
56 /*
57 * Detect falling edge: was HIGH (released), now LOW (pressed)
58 * Active-low buttons: bit set = released, bit clear = pressed
59 */
60
61 /* Reset button (PD3) */
62 if ((g_btn_prev & (1 << BTN_RESET_PIN)) && !(btn_state & (1 << BTN_RESET_PIN))) {
63 events |= BTN_RESET_PRESSED;
64 }
65
66 /* Preset button (PD4) */
67 if ((g_btn_prev & (1 << BTN_PRESET_PIN)) && !(btn_state & (1 << BTN_PRESET_PIN))) {
68 events |= BTN_PRESET_PRESSED;
69 }
70
71 /* Start button (PD5) */
72 if ((g_btn_prev & (1 << BTN_START_PIN)) && !(btn_state & (1 << BTN_START_PIN))) {
73 events |= BTN_START_PRESSED;
74 }
75
76 /* Update state for next comparison */
77 g_btn_prev = btn_state;
78
79 /* Reset debounce timer on any button press */
80 if (events) {
81 g_last_time = now;
82 }
83
84 return events;
85}
